# 3-D cross-section weight w(Q2, W, M_X), built by
# reweight/build_xsec_weight3d.py. M_X is the invariant mass of the
# intermediate from the first vertex -- M_ppbar for the reaction above.
# Applied as a THIRD accept-reject, after the decay chain, since M_X does
# not exist at electron-sampling time.
#   Format:  xsec_weight: <root file> [<hist name>]   (default w_Q2_W_M)
#            xsec_weight_mode: bin | interp           (default bin)
# `bin` looks up the bin the event falls in -- the right pairing for a
# BINNED cross section. `interp` trilinearly interpolates and, on a coarse
# grid, pulls the result well away from the cross section it was built from.
# xsec_weight: xsec_weight.root w_Q2_W_M
# xsec_weight_mode: bin

# Per-accepted-event truth TTree (Q2, W, M, Ep, theta_e, w_pbarp, w_pp,

# build_xsec_weight3d.py (the 3-D proposal is not flat), read by

truth_ntuple: gen_truth.root

# Rescattering weights. The one ep -> e'p p pbar final state holds BOTH
# rescattering subsystems, so every event gets two weights,
#   w_pbarp = sigma_pbarp(s_rpbar, t) / D_gen(s_rpbar, t)
#   w_pp    = sigma_pp   (s_rp,    t) / D_gen(s_rp,    t)
# and a FOURTH accept-reject keeps it with probability
# (w_pbarp * w_pp) / w_max. The single sample that comes out is
# unweighted and distributed as D_gen x w_pbarp x w_pp -- the generated
# distribution multiplied by both weights, each factor reshaping its own
# subsystem. One LUND file, the full reaction, as in an unweighted run.
#
# Each model is a TFormula in s and t, or a TH2D table from
# reweight/build_dsdt_table.py (prefer --log + ratio_weight_mode: log).
# D_gen is the generated (s, t) density from an unweighted run
# (build_dsdt_table.py --from-gen); optional, and what makes each factor
# reproduce its sigma rather than sigma x D_gen.
#
# ratio_weight_max is the accept-reject ceiling on the product: any
# number at or above max(w_pbarp * w_pp), a plain constant included. Too
# high only costs efficiency. Omit it and the ceiling is scanned off the
# tables and the D_gen grid at load time -- but a formula model with
# neither has no grid to scan, so there it is required. Events whose
# product exceeds the ceiling are clamped and counted in the run summary.
#
# ratio_weight_apply: carry turns the selection off instead: every event
# kept, both weights carried (truth branches w_pbarp, w_pp, w_ratio,
# w_event; optional two-column sidecar).
ratio_weight_formula:     exp(4.0*t+3)
